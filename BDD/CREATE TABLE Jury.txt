# 1. CREATION DES TABLES

CREATE TABLE jury(
   id_Jury INT,
   president INT,
   PRIMARY KEY(id_Jury)
);

CREATE TABLE publisher(
   id_publisher INT,
   name VARCHAR(50),
   PRIMARY KEY(id_publisher)
);

CREATE TABLE selection(
   id_selection INT,
   date_selection DATE,
   round INT,
   PRIMARY KEY(id_selection)
);

CREATE TABLE person(
   id_person INT,
   first_name VARCHAR(50),
   last_name VARCHAR(50),
   PRIMARY KEY(id_person)
);

CREATE TABLE user_(
   id_user INT,
   user_name VARCHAR(50),
   password VARCHAR(50),
   id_person INT NOT NULL,
   PRIMARY KEY(id_user),
   UNIQUE(id_person),
   FOREIGN KEY(id_person) REFERENCES Person(id_person)
);

CREATE TABLE author(
   id_author INT,
   biographie TEXT,
   id_person INT,
   PRIMARY KEY(id_author),
   UNIQUE(id_person),
   FOREIGN KEY(id_person) REFERENCES Person(id_person)
);

CREATE TABLE book(
   isbn BIGINT,
   title VARCHAR(100) NOT NULL,
   summary TEXT,
   publication_date DATE,
   number_pages INT,
   publisher_price DECIMAL(10,2),
   id_publisher INT NOT NULL,
   id_author INT NOT NULL,
   PRIMARY KEY(isbn),
   FOREIGN KEY(id_publisher) REFERENCES Publisher(id_publisher),
   FOREIGN KEY(id_author) REFERENCES Author(id_author)
);

CREATE TABLE character_(
   id_character INT,
   name VARCHAR(50) NOT NULL,
   isbn BIGINT NOT NULL,
   PRIMARY KEY(id_character),
   FOREIGN KEY(isbn) REFERENCES Book(isbn)
);

CREATE TABLE is_selected(
   isbn BIGINT,
   id_selection INT,
   PRIMARY KEY(isbn, id_selection),
   FOREIGN KEY(isbn) REFERENCES Book(isbn),
   FOREIGN KEY(id_selection) REFERENCES Selection(id_selection)
);

CREATE TABLE is_member(
   id_Jury INT,
   id_person INT,
   PRIMARY KEY(id_Jury, id_person),
   FOREIGN KEY(id_Jury) REFERENCES Jury(id_Jury),
   FOREIGN KEY(id_person) REFERENCES Person(id_person)
);

CREATE TABLE vote(
   isbn BIGINT,
   id_Jury INT,
   id_selection INT,
   number_of_votes INT,
   PRIMARY KEY(isbn, id_Jury, id_selection),
   FOREIGN KEY(isbn) REFERENCES Book(isbn),
   FOREIGN KEY(id_Jury) REFERENCES Jury(id_Jury),
   FOREIGN KEY(id_selection) REFERENCES Selection(id_selection)
);
